#!/bin/sh
# .git を自分自身へ bind（pin）すると、mv で差し替えられなくなるかの確認。
set -u
T="${TMPDIR:-/tmp}/mvgit-test2"
rm -rf "$T"
mkdir -p "$T/repo/.git/hooks" "$T/repo/.git/objects"
echo "[core]" > "$T/repo/.git/config"
echo "HEAD" > "$T/repo/.git/HEAD"
echo "x" > "$T/repo/.git/commondir"
if ! command -v bwrap >/dev/null 2>&1; then echo "SKIP: bwrap が無い"; exit 0; fi
bwrap --die-with-parent --unshare-all --dev /dev --proc /proc --ro-bind /usr /usr --symlink usr/bin /bin --symlink usr/lib /lib --symlink usr/lib64 /lib64 \
  --bind "$T/repo" /w \
  --bind "$T/repo/.git" /w/.git \
  --ro-bind "$T/repo/.git/config" /w/.git/config \
  --ro-bind "$T/repo/.git/hooks" /w/.git/hooks \
  --ro-bind "$T/repo/.git/commondir" /w/.git/commondir \
  --chdir /w \
  /bin/sh -c '
    echo "-- mv .git .git.old (with pin):"; mv .git .git.old 2>&1; echo "mv_rc=$?"
    echo "-- rm -rf .git (with pin):"; rm -rf .git 2>&1 | head -2; echo "rm_done; .git exists? $( [ -d .git ] && echo yes || echo no )"
    echo "-- ro still holds:"; (echo evil >> .git/config) 2>&1 | head -1
    echo "-- ordinary git writes still work (objects/HEAD):"; echo ok > .git/objects/x && echo objects_write_ok; echo HEAD2 > .git/HEAD && echo head_write_ok
  '
echo "-- host config after:"; cat "$T/repo/.git/config"
